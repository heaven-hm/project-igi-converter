cmake_minimum_required(VERSION 3.16)
project(igi_converters VERSION 1.11.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(IGICONV_BUILD_GUI "Build both converters with the shared Qt desktop GUI" ON)
if(IGICONV_BUILD_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets Multimedia)
endif()
file(GLOB parsers CONFIGURE_DEPENDS source/parsers/*.cpp)
list(FILTER parsers EXCLUDE REGEX "/(qsc_lexer|qsc_parser|qvm_compiler)\\.cpp$")
file(GLOB codecs CONFIGURE_DEPENDS native/*.cpp)
list(FILTER codecs EXCLUDE REGEX "/(dispatch|extended|qsc)\\.cpp$")
# The format readers, codecs and desktop UI are shared. Each executable has
# its own command adapter and script dialect; both targets are built together.
foreach(game IN ITEMS 1 2)
  set(target igi${game}conv)
  if(game EQUAL 1)
    file(GLOB commands CONFIGURE_DEPENDS igi1conv/*.cpp)
    list(APPEND commands native/dispatch.cpp source/parsers/qsc_lexer.cpp source/parsers/qsc_parser.cpp source/parsers/qvm_compiler.cpp)
  else()
    file(GLOB commands CONFIGURE_DEPENDS igi2conv/*.cpp)
    list(APPEND commands native/extended.cpp native/qsc.cpp native/qsc/qsc_lexer.cpp native/qsc/qsc_parser.cpp native/qsc/qvm_compiler.cpp)
  endif()
  add_executable(${target} ${commands} ${parsers} ${codecs})
  target_include_directories(${target} PRIVATE gui ${target} source source/parsers native third_party/glm-1.0.1 third_party)
  target_compile_definitions(${target} PRIVATE CONVERTER_GAME=${game} IGI1CONV_VERSION="1.11.0")
  if(IGICONV_BUILD_GUI)
    target_sources(${target} PRIVATE gui/gui_main.cpp gui/gui_main.h assets/resources.qrc)
    set_target_properties(${target} PROPERTIES AUTOMOC ON AUTORCC ON)
    target_link_libraries(${target} PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::OpenGL Qt6::OpenGLWidgets Qt6::Multimedia)
    target_compile_definitions(${target} PRIVATE IGI1CONV_WITH_GUI=1 IGICONV_WITH_GUI=1)
    if(WIN32)
      target_link_libraries(${target} PRIVATE winmm)
    endif()
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /bigobj /FS)
    target_compile_definitions(${target} PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  endif()
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/IGIModels.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/IGIModelsAllLevel.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/IGIAutoComplete.txt"
    $<TARGET_FILE_DIR:${target}>)
  install(TARGETS ${target} RUNTIME DESTINATION bin)
endforeach()
