cmake_minimum_required(VERSION 3.16)
project(igi2conv VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(GLOB native_sources CONFIGURE_DEPENDS "macos-gconv/src/*.cpp")
list(APPEND native_sources macos-gconv/src/qsc/qsc_lexer.cpp
  macos-gconv/src/qsc/qsc_parser.cpp macos-gconv/src/qsc/qvm_compiler.cpp)
file(GLOB parser_sources CONFIGURE_DEPENDS "source/parsers/*.cpp")
list(FILTER parser_sources EXCLUDE REGEX "/(qsc_lexer|qsc_parser|qvm_compiler)\\.cpp$")
set(extra_commands mef qvm mtp terrain graph dat iff wav olm lightmap tex)
set(extra_sources igi1conv/support.cpp)
foreach(command IN LISTS extra_commands)
  list(APPEND extra_sources "igi1conv/cmd_${command}.cpp")
endforeach()
add_executable(igi2conv ${native_sources} ${parser_sources} ${extra_sources})
target_include_directories(igi2conv PRIVATE igi1conv source source/parsers third_party/glm-1.0.1 third_party)

if(MSVC)
  target_compile_options(igi2conv PRIVATE /W4 /bigobj /FS)
  set_property(TARGET igi2conv PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_definitions(igi2conv PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(igi2conv PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(IGICONV_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(IGICONV_ENABLE_SANITIZERS AND NOT MSVC)
  target_compile_options(igi2conv PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(igi2conv PRIVATE -fsanitize=address,undefined)
endif()
install(TARGETS igi2conv RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME igi2conv_cli COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/macos-gconv/verification/check_cli.py
    --binary $<TARGET_FILE:igi2conv> --report ${CMAKE_CURRENT_BINARY_DIR}/cli-results.json)
endif()

if(MINGW)
  target_link_options(igi2conv PRIVATE -static -static-libgcc -static-libstdc++)
endif()
